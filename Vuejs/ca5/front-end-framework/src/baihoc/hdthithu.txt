5. ROUTER

import { createRouter, createWebHistory } from "vue-router"

import StudentList from "../views/StudentList.vue"
import StudentForm from "../views/StudentForm.vue"
import StudentDetail from "../views/StudentDetail.vue"

const routes = [
  { path: "/", component: StudentList },
  { path: "/add", component: StudentForm },
  { path: "/edit/:id", component: StudentForm },
  { path: "/students/:id", component: StudentDetail }
]

export default createRouter({
  history: createWebHistory(),
  routes
})

6. CHỨC NĂNG 1 — HIỂN THỊ DANH SÁCH: StudentList.vue
<script setup>
import { ref, onMounted } from "vue"
import api from "../services/api"
import { useRouter } from "vue-router"

const students = ref([])
const router = useRouter()

const fetchStudents = async () => {
  const res = await api.get("/students")
  students.value = res.data
}

const goDetail = (id) => {
  router.push(`/students/${id}`)
}

const goEdit = (id) => {
  router.push(`/edit/${id}`)
}

const removeStudent = async (id) => {
  await api.delete(`/students/${id}`)
  fetchStudents()
}

onMounted(fetchStudents)
</script>

<template>
  <h1>Danh sách sinh viên</h1>

  <button @click="router.push('/add')">Thêm</button>

  <ul>
    <li v-for="s in students" :key="s.id">
      {{ s.name }} - {{ s.age }}

      <button @click="goDetail(s.id)">Xem</button>
      <button @click="goEdit(s.id)">Sửa</button>
      <button @click="removeStudent(s.id)">Xóa</button>
    </li>
  </ul>
</template>

7. CHỨC NĂNG 2 — THÊM & SỬA: StudentForm.vue
<script setup>
import { ref, onMounted } from "vue"
import api from "../services/api"
import { useRoute, useRouter } from "vue-router"

const route = useRoute()
const router = useRouter()

const student = ref({
  name: "",
  age: "",
  email: "",
  avatar: ""
})

const isEdit = route.params.id

// load data nếu edit
onMounted(async () => {
  if (isEdit) {
    const res = await api.get(`/students/${isEdit}`)
    student.value = res.data
  }
})

// submit
const submit = async () => {
  if (isEdit) {
    await api.patch(`/students/${isEdit}`, student.value)
  } else {
    await api.post("/students", student.value)
  }

  router.push("/")
}
</script>

<template>
  <h1>{{ isEdit ? "Sửa" : "Thêm" }} sinh viên</h1>

  <input v-model="student.name" placeholder="Tên" />
  <input v-model="student.age" placeholder="Tuổi" />
  <input v-model="student.email" placeholder="Email" />
  <input v-model="student.avatar" placeholder="Avatar" />

  <button @click="submit">Lưu</button>
</template>

8. CHỨC NĂNG 3 — CHI TIẾT: StudentDetail.vue
<script setup>
import { ref, onMounted } from "vue"
import api from "../services/api"
import { useRoute } from "vue-router"

const route = useRoute()
const student = ref(null)

onMounted(async () => {
  const res = await api.get(`/students/${route.params.id}`)
  student.value = res.data
})
</script>

<template>
  <div v-if="student">
    <h2>{{ student.name }}</h2>
    <p>Age: {{ student.age }}</p>
    <p>Email: {{ student.email }}</p>
    <img :src="student.avatar" width="100" />
  </div>
</template>
